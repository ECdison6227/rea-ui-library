通用聊天卡片结构（虚构示例）TypeScript
// 用 kind 做判别字段，前端 switch(kind) 选组件
type Card =
  | { kind: "linkStack"; links: { href: string; title: string; thumb?: string; tag?: string }[] }
  | { kind: "gallery"; photos: { src: string; alt: string }[] }
  | { kind: "grid"; heading: string; headers: string[]; cells: string[][]; imageCol?: number }
  | { kind: "doc"; heading: string; md: string; collapsedLines?: number }
  | { kind: "checklist"; heading: string; entries: { text: string; done: boolean; due?: string }[] }
  | { kind: "results"; query: string; hits: { href: string; title: string; snippet: string }[] }
  | { kind: "snippet"; lang: string; heading: string; body: string }
  | { kind: "receipt"; vendor: string; rows: { label: string; amount: string; qty?: number }[]; sum: string; cta?: string }
  | { kind: "sitePreview"; heading: string; frameSrc: string }
  | { kind: "attachment"; filename: string; mime: string; size: number; previewSrc?: string };

// 每张卡片的生命周期，驱动骨架屏和展开/收起
type CardState = "pending" | "ready" | "error";
interface CardEnvelope { id: string; state: CardState; expanded: boolean; card?: Card }

// 虚构样例
const demo: CardEnvelope[] = [
  { id: "c1", state: "ready", expanded: false,
    card: { kind: "linkStack", links: [
      { href: "https://example.com/a", title: "示例咖啡馆", tag: "菜单" },
      { href: "https://example.com/b", title: "示例书店", tag: "地图" } ] } },
  { id: "c2", state: "pending", expanded: false },  // 先显示骨架，数据到了再切 ready
  { id: "c3", state: "ready", expanded: false,
    card: { kind: "doc", heading: "虚构周末计划", md: "## 周六\n- 10:00 公园散步", collapsedLines: 6 } }
];

// 渲染要点（通用建议）：
// - linkStack / gallery 超过 1 项时叠成一叠，点开再横向铺开
// - doc 超过 collapsedLines 行时折叠，点击切换 expanded
// - pending 显示骨架屏，error 显示重试按钮，ready 时淡入
